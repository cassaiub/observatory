\begin{frame}{Phase 1 --- seeding the error budget \asidetag}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      Poisson (photon shot noise) plus read noise, on the counts the detector
      \emph{collected} --- raw minus the bias pedestal $B$:
      \[
        \sigma_{\text{ADU}} = \frac{\sqrt{g \cdot (I - B) + \text{RN}^2}}{g}
      \]
      {\footnotesize Treating the pedestal as signal would inject tens of
      fictitious electrons into \emph{every} pixel.}
      Subtracting a master adds in quadrature:
      \[
        \sigma_{\text{out}} = \sqrt{\sigma_{\text{sci}}^2 + \sigma_{\text{bias}}^2}
      \]
      Dividing by the flat propagates \emph{relative} errors:
      \[
        \left(\frac{\sigma_{\text{out}}}{S_{\text{out}}}\right)^2 =
        \left(\frac{\sigma_{\text{sci}}}{S}\right)^2 +
        \left(\frac{\sigma_{\text{flat}}}{F}\right)^2
      \]
    \end{column}
    \begin{column}{0.46\textwidth}
      \centering
      \begin{tikzpicture}[node distance=2mm]
        \node[errbox, text width=3.4cm] (a) {seed: $\sqrt{g(I-B)+\mathrm{RN}^2}/g$};
        \node[errbox, below=3mm of a, text width=3.4cm] (b) {$\oplus\;\sigma_{\text{bias}}$};
        \node[errbox, below=3mm of b, text width=3.4cm] (c) {$\oplus\;\sigma_{\text{dark}}$};
        \node[errbox, below=3mm of c, text width=3.4cm] (d) {$\oplus\;\sigma_{\text{flat}}/F$ \tiny(relative)};

\begin{frame}{Phase 2 --- why inverse-variance weighting? \asidetag}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      Weight frame $i$ by $w_i = 1/\mathrm{var}_i$:
      \[
        \text{master} = \frac{\sum_i w_i\,\text{sci}_i}{\sum_i w_i},
        \qquad
        \mathrm{var}_{\text{master}} = \frac{1}{\sum_i w_i}.
      \]
      \begin{itemize}\footnotesize
        \item Noisy frames contribute less; this is the \emph{optimal} linear combination.
        \item $N$ equal frames $\Rightarrow$ noise shrinks by $\sqrt{N}$.
        \item Rejection: sigma-clip at $\ge5$ frames, min/max below.
      \end{itemize}
    \end{column}
    \begin{column}{0.43\textwidth}
      \centering
      \begin{tikzpicture}[node distance=2mm, scale=0.9, transform shape]
        \node[dat, text width=0.9cm] (f1) {$w_1$};
        \node[dat, right=2mm of f1, text width=0.9cm] (f2) {$w_2$};
        \node[dat, right=2mm of f2, text width=0.9cm] (f3) {$\cdots$};
        \node[dat, right=2mm of f3, text width=0.9cm] (f4) {$w_N$};
        \node[op, below=7mm of f2, text width=3.0cm, xshift=8mm] (st) {weighted mean};
        \node[prod, below=5mm of st, text width=3.0cm] (m) {deep master\\ \tiny noise $/\sqrt{N}$};
        \foreach \x in {f1,f2,f3,f4} \draw[garr] (\x) -- (st);
        \draw[arr] (st) -- (m);
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Phase 3 --- the zero point, with an error bar \asidetag}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      For each catalog-matched star:
      \[
        m_{\text{inst}} = -2.5\log_{10}(F), \quad
        \delta m_{\text{inst}} \approx 1.0857\,\frac{\delta F}{F}
      \]
      \[
        \mathrm{ZP}_i = m_{\text{cat}} - m_{\text{inst}}, \quad
        \delta \mathrm{ZP}_i = \sqrt{\delta m_{\text{inst}}^2 + \delta m_{\text{cat}}^2}
      \]
      Combine with sigma-clipping and $w_i = 1/\delta\mathrm{ZP}_i^2$; report
      \texttt{MAGZERR} as the \emph{larger} of the formal error and the scatter's
      standard error. Final source magnitude:
      \[
        \delta m = \sqrt{(1.0857\,\delta F/F)^2 + \texttt{MAGZERR}^2}
      \]
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \begin{tikzpicture}[node distance=2.5mm, scale=0.92, transform shape]
        \node[dat, text width=3.2cm] (a) {$N$ matched stars};
        \node[op, below=4mm of a, text width=3.2cm] (b) {per-star $\mathrm{ZP}_i \pm \delta\mathrm{ZP}_i$};
        \node[op, below=4mm of b, text width=3.2cm] (c) {sigma-clip outliers};
        \node[op, below=4mm of c, text width=3.2cm] (d) {inverse-variance mean};
        \node[prod, below=4mm of d, text width=3.2cm] (e) {\texttt{MAGZERO} $\pm$ \texttt{MAGZERR}};
        \foreach \x/\y in {a/b, b/c, c/d, d/e} \draw[arr] (\x) -- (\y);
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Which magnitude? \asidetag}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      An aperture never catches \emph{all} of a star's light, and how much it
      misses depends on the source.

      \medskip
      \texttt{MAG\_ISO} is \textbf{isophotal} --- the flux above the detection
      threshold. A bright star's footprint reaches further down its wings than a
      faint one's, so the fraction captured \emph{depends on brightness}.

      \medskip
      Applying an aperture-derived zero point to it therefore \textbf{tilts} the
      magnitude scale. A tilt is not an offset: no single correction removes it.

      \medskip
      {\footnotesize Measured against simulated data with known truth.}
    \end{column}
    \begin{column}{0.46\textwidth}
      \centering
      {\footnotesize
      \begin{tabular}{lrr}
        \toprule
        column & median & trend \\
        \midrule
        \texttt{MAG\_BEST} & $+0.03$ & $+0.004$ \\
        \texttt{MAG\_APER} & $-0.01$ & $-0.011$ \\
        \texttt{MAG\_ISO}  & $+0.53$ & $\mathbf{+0.348}$ \\
        \bottomrule
      \end{tabular}
      \par\medskip
      trend in mag per mag
      }

      \vspace{1em}
      \begin{block}{\small Which column, then?}
        \footnotesize \texttt{MAG\_BEST} --- PSF for point sources, Kron for extended --- is
        the \emph{total} magnitude. \texttt{MAG\_APER} is measured in the same aperture the
        zero point was, so it is the one exactly consistent with the calibration.
        \texttt{MAG\_ISO} is kept only for comparison with older catalogs.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Star or galaxy? Concentration, not shape \asidetag}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      The discriminant is $\texttt{MAG\_PSF} - \texttt{MAG\_AUTO}$: how much of a
      source's light lies outside the PSF core.

      \medskip
      \textbf{Not ellipticity.} That is \emph{shape}: a face-on elliptical
      galaxy is round and passes as a star; a slightly trailed star is elongated
      and fails.

      \medskip
      The threshold is calibrated \textbf{per frame} from the stellar locus
      itself --- so it adapts to tonight's seeing, with no training data.
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{block}{\small \texttt{AMBIGUOUS} is an answer}
        \footnotesize Below the SNR where the locus separation falls under its
        own scatter, no class is asserted. \texttt{CLASSLIM} records where that
        happens, rather than pretending to classify to the detection limit.
      \end{block}
      \medskip
      {\footnotesize Why it matters beyond the catalog column: a galaxy leaking
      into the zero point biases \emph{every} magnitude in the frame.}
    \end{column}
  \end{columns}
\end{frame}
